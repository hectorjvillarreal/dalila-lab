% ---------- 10. Frente federal: impuesto mínimo ----------
\begin{frame}{El paquete ya trae un impuesto mínimo; el diseño importa más que el nombre}
  \setbeamercolor{bloque}{bg=fondo,fg=gris}
  \begin{beamercolorbox}[wd=\linewidth,sep=7pt]{bloque}
    \small\punto El tope a deducciones de 96.67\,\% equivale a un ISR mínimo de alrededor de 1\,\% de los ingresos para empresas con ingresos mayores a 50\,mdp.\par\vspace{0.4em}
    \punto Un impuesto de $t$ sobre ingresos equivale a un piso de margen de $t/\tau$: con 1\,\% y una tasa de 30\,\%, el piso es 3.33\,\%. Las empresas con márgenes legítimamente bajos pagan de más.\par\vspace{0.4em}
    \punto Un mínimo «acreditable» tiende a volverse definitivo en la práctica: en Pakistán, solo 1.3\,\% de las empresas usa el crédito.
  \end{beamercolorbox}
  \vspace{0.5em}
  \setbeamercolor{banda}{bg=petroleo,fg=white}
  \begin{beamercolorbox}[wd=\textwidth,sep=6pt,center]{banda}
    \bfseries Lo pendiente en el frente federal: los gastos fiscales.
  \end{beamercolorbox}
  \fuente{\raggedright Minuta de reforma a la LISR, Cámara de Diputados, octubre de 2026. Pakistán: Best, Brockmeyer, Kleven, Spinnewijn y Waseem (2015), \emph{Journal of Political Economy} 123(6).\par}
\end{frame}

% ---------- 12. Seguridad social (I): Modalidad 40 ----------
\begin{frame}{Modalidad 40: lo que cambió el 21 de octubre}
  \setbeamercolor{bloque}{bg=fondo,fg=gris}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \frente{Hasta el 21 de octubre}{\punto Cualquier salario de cotización de hasta 25 UMA (casi 3 mil pesos diarios), sin relación con el salario previo.\par\vspace{0.5em}\punto Se permitían inscripciones con fecha anterior.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \frente{Desde el 22 de octubre}{\punto El último salario registrado, o ese salario más una UMA (unos 117 pesos diarios), nunca por debajo del mínimo.\par\vspace{0.5em}\punto Sin inscripciones retroactivas; trámite solo digital.\par\vspace{0.5em}\punto Los inscritos conservan sus condiciones hasta terminar su periodo.}
    \end{column}
  \end{columns}
  \fuente{\raggedright Fuente: Acuerdo ACDO.AS2.HCT.101026/302.P.DG, \emph{Diario Oficial de la Federación}, 21 de octubre de 2026.\par}
\end{frame}

% ---------- 13. Seguridad social (II): quién paga ----------
\begin{frame}{Seguir cotizando es razonable; comprar una pensión no lo es}
  \setbeamercolor{bloque}{bg=fondo,fg=gris}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      \frente{Quién paga}{\footnotesize\punto Para quien cotizó antes de julio de 1997, la pensión usa el salario promedio de los últimos cinco años: la estrategia abría la puerta a pensiones vitalicias de más de 80 mil pesos al mes.\par\vspace{0.2em}\punto Las cuotas no alcanzan; la diferencia la paga el Gobierno Federal, es decir, todos los contribuyentes, incluidos quienes nunca tendrán pensión.\par\vspace{0.2em}\punto Solo está al alcance de quien puede adelantar cientos de miles de pesos.}
    \end{column}
    \begin{column}{0.46\textwidth}
      \frente{Lo que falta}{\footnotesize\punto Es una medida temporal: seis meses para estimar el costo y otros seis para proponer cambios a las normas.\par\vspace{0.2em}\punto El análisis de costo debe hacerse público.\par\vspace{0.2em}\punto La solución definitiva necesita base legal firme: un acuerdo invita a los amparos.\par\vspace{0.2em}\punto Debe respetarse a quien ya cotiza.}
    \end{column}
  \end{columns}
  \fuente{\raggedright Fuente: Acuerdo ACDO.AS2.HCT.101026/302.P.DG (DOF, 21-10-2026); Ley del Seguro Social de 1973.\par}
\end{frame}
